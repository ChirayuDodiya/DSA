## Merge Sort

### Approach

Divide the array into smaller parts, sort each part, then merge the sorted parts in order. Return the final sorted array.

### Complexity

- **Time:** `O(n log n)`, because the array is split across `log n` levels and each level performs `O(n)` total merging work.
- **Auxiliary space:** `O(n)` for temporary merge arrays and `O(log n)` for the recursion stack in balanced recursive calls.

### Implementation

```cpp
#include <bits/stdc++.h>
using namespace std;

class Solution {
private:
    void merge(vector<int>& arr, int left, int mid, int right) {
        vector<int> leftPart;
        vector<int> rightPart;

        for (int index = left; index <= mid; index++) {
            leftPart.push_back(arr[index]);
        }

        for (int index = mid + 1; index <= right; index++) {
            rightPart.push_back(arr[index]);
        }

        int leftIndex = 0;
        int rightIndex = 0;
        int writeIndex = left;

        while (leftIndex < leftPart.size() && rightIndex < rightPart.size()) {
            if (leftPart[leftIndex] <= rightPart[rightIndex]) {
                arr[writeIndex] = leftPart[leftIndex];
                leftIndex++;
            } else {
                arr[writeIndex] = rightPart[rightIndex];
                rightIndex++;
            }
            writeIndex++;
        }

        while (leftIndex < leftPart.size()) {
            arr[writeIndex] = leftPart[leftIndex];
            leftIndex++;
            writeIndex++;
        }

        while (rightIndex < rightPart.size()) {
            arr[writeIndex] = rightPart[rightIndex];
            rightIndex++;
            writeIndex++;
        }
    }

    void mergeSortHelper(vector<int>& arr, int left, int right) {
        if (left >= right) {
            return;
        }

        int mid = left + (right - left) / 2;

        mergeSortHelper(arr, left, mid);

        mergeSortHelper(arr, mid + 1, right);

        merge(arr, left, mid, right);
    }

public:
    vector<int> mergeSort(vector<int>& arr) {
        int n = arr.size();

        if (n < 2) {
            return arr;
        }

        mergeSortHelper(arr, 0, n - 1);
        return arr;
    }
};
```
